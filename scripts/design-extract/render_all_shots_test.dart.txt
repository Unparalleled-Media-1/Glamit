Future<void> font(String family, List<String> files) async {
  final l = FontLoader(family);
  for (final f in files) { l.addFont(Future.value(ByteData.view(File(f).readAsBytesSync().buffer))); }
  await l.load();
}
void main() {
  testWidgets('all shots', (tester) async {
    final theme = const String.fromEnvironment('THEME', defaultValue: 'dark'), plat = const String.fromEnvironment('PLAT', defaultValue: 'ios'), out = const String.fromEnvironment('OUT');
    await font('Arial', ['/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf']);
    await font('Georgia', ['/usr/share/fonts/truetype/dejavu/DejaVuSerif.ttf']);
    await font('GlamSymbols', ['assets/fonts/DejaVuSans.ttf']);
    tester.view.physicalSize = const Size(390, 844); tester.view.devicePixelRatio = 1;
    addTearDown(tester.view.resetPhysicalSize); addTearDown(tester.view.resetDevicePixelRatio);
    debugDefaultTargetPlatformOverride = plat == 'android' ? TargetPlatform.android : TargetPlatform.iOS;
    try {
      final v = '${theme}_$plat';
      final data = await tester.runAsync(() => DesignStore.load(v)) as Map<String, dynamic>;
      Directory(out).createSync(recursive: true);
      for (final id in data.keys) {
        final key = GlobalKey();
        await tester.pumpWidget(MaterialApp(debugShowCheckedModeBanner: false, theme: MobileDesign.theme(dark: theme == 'dark'), home: RepaintBoundary(key: key, child: Scaffold(body: DesignScreen(id: id, go: (_) {})))));
        await tester.pump();
        await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 60)));
        await tester.pump(const Duration(milliseconds: 50));
        await tester.runAsync(() async {
          final img = await (key.currentContext!.findRenderObject() as RenderRepaintBoundary).toImage(pixelRatio: 1);
          File('$out/flutter_$id.png').writeAsBytesSync((await img.toByteData(format: ui.ImageByteFormat.png))!.buffer.asUint8List());
        });
      }
    } finally { debugDefaultTargetPlatformOverride = null; }
  }, timeout: const Timeout(Duration(minutes: 20)));
}
